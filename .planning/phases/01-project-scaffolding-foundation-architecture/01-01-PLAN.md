<task type="auto">
  <name>Task 2: Implement Baseline Flyway Migration for Core PostgreSQL Tables</name>
  <files>

  <action>
    Create backend/src/main/resources/db/migration/V1__init_schema.sql per D-06 defining the 9 core tables from SKILLSWAP_SPEC.md § 9:
    1. users (id BIGSERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(150) UNIQUE NOT NULL, password_hash VARCHAR(255) NOT NULL, bio TEXT, location VARCHAR(100), role VARCHAR(50) DEFAULT 'ROLE_MEMBER', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    2. skills (id BIGSERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, category VARCHAR(100) NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    3. user_skills (id BIGSERIAL PRIMARY KEY, user_id BIGINT REFERENCES users(id) ON DELETE CASCADE, skill_id BIGINT REFERENCES skills(id) ON DELETE CASCADE, type VARCHAR(20) NOT NULL, experience VARCHAR(50), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    4. wallets (id BIGSERIAL PRIMARY KEY, user_id BIGINT UNIQUE REFERENCES users(id) ON DELETE CASCADE, balance NUMERIC(10, 2) NOT NULL DEFAULT 5.00, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    5. exchange_requests (id BIGSERIAL PRIMARY KEY, requester_id BIGINT REFERENCES users(id), provider_id BIGINT REFERENCES users(id), skill_id BIGINT REFERENCES skills(id), duration NUMERIC(5, 2) NOT NULL, status VARCHAR(50) NOT NULL DEFAULT 'PENDING', notes TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    6. bookings (id BIGSERIAL PRIMARY KEY, request_id BIGINT REFERENCES exchange_requests(id) ON DELETE CASCADE, start_time TIMESTAMP NOT NULL, end_time TIMESTAMP NOT NULL, status VARCHAR(50) NOT NULL DEFAULT 'SCHEDULED', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    7. credit_transactions (id BIGSERIAL PRIMARY KEY, exchange_id BIGINT REFERENCES exchange_requests(id), sender_id BIGINT REFERENCES users(id), receiver_id BIGINT REFERENCES users(id), amount NUMERIC(10, 2) NOT NULL, type VARCHAR(50) NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    8. reviews (id BIGSERIAL PRIMARY KEY, exchange_id BIGINT REFERENCES exchange_requests(id), reviewer_id BIGINT REFERENCES users(id), reviewee_id BIGINT REFERENCES users(id), rating INT CHECK (rating >= 1 AND rating <= 5), comment TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    9. notifications (id BIGSERIAL PRIMARY KEY, user_id BIGINT REFERENCES users(id) ON DELETE CASCADE, message TEXT NOT NULL, type VARCHAR(50) NOT NULL, is_read BOOLEAN DEFAULT FALSE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)
    Create SkillSwapApplicationTests.java to verify Spring Boot application context loads with the database schema.
  </action>

  </verify>
